async function validar() { const contrasena = document.getElementById('password').value; const resultado = document.getElementById('resultado'); const titulo = document.getElementById('titulo-resultado'); const lista = document.getElementById('lista-reglas'); if (!contrasena) return; titulo.textContent = 'Validando...'; resultado.style.display = 'block'; resultado.className = 'resultado'; try { const response = await fetch('https://twodainstanciaevaluativa.onrender.com/validar', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contrasena: contrasena }) }); const data = await response.json(); resultado.className = 'resultado ' + (data.valida ? 'valida' : 'invalida'); titulo.textContent = data.valida ? '✅ Contraseña segura' : '❌ Contraseña insegura'; const reglaTextos = { 'Debe tener al menos 8 caracteres': 'Al menos 8 caracteres', 'Debe tener al menos una mayuscula': 'Al menos una letra mayúscula', 'Debe tener al menos un numero': 'Al menos un número', 'Debe tener al menos un simbolo especial': 'Al menos un símbolo especial (!@#$%...)' }; const todasLasReglas = Object.values(reglaTextos); lista.innerHTML = todasLasReglas.map(regla => { const estaEnErrores = data.errores.some(e => reglaTextos[e] === regla); return `
  • ${estaEnErrores ? '✖' : '✔'} ${regla}
    • `; }).join(''); } catch (error) { titulo.textContent = '⚠️ Error conectando con el servidor'; resultado.className = 'resultado invalida'; } }
